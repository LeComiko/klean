# Version 0.3.0 — Le géocodage des sites — Plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal :** remplir la latitude et la longitude d'un site depuis son adresse, par la Géoplateforme de
l'IGN, sur un clic « Localiser », derrière un réglage de l'entreprise désactivé par défaut.

**Architecture :** un réglage `Entreprise.GeocodageActif` et ses deux dates ; une route
`POST /api/sites/localiser`, administrateur seul, plafonnée par compte et globalement ; un service
concret `ServiceGeocodage` sur un client HTTP nommé, verrouillé (pas de redirection, de cookie, de
`traceparent`, de journal), avec un disjoncteur sur les 429 ; côté écran, une propriété déclarative
`localiser` sur la description des sites, rendue par `FormulaireRessource`, orchestrée par
`Reference.tsx`, la logique dans `reference/localiser.ts`.

**Tech Stack :** ASP.NET Core 10, EF Core 10 / Npgsql, `IHttpClientFactory`, xUnit ; React 19 +
TypeScript, `node --test`.

**La spécification** : `docs/superpowers/specs/2026-11-23-geocodage-design.md`. La lire avant toute
tâche. Ce plan la suit ; là où il précise (les tests du service écrits au niveau du service plutôt que
sur un hôte dérivé), c'est dit.

**Fichiers :** créer `backend/Services/ServiceGeocodage.cs` ; modifier `backend/Program.cs`.

- [ ] **Étape 1 : écrire le service entier** (la tâche 5 l'éprouve ; il est écrit d'un tenant parce
  que ses parties se tiennent) :

```csharp
using System.Diagnostics;
using System.Globalization;
using System.Net;
using System.Text;
using System.Text.Json;

namespace Klean.Api.Services;

/// <summary>Ce que rend une localisation. Aucune autre issue : l'appelant n'a pas à lire d'exception.</summary>
public abstract record ResultatGeocodage
{
    public sealed record Trouve(double Latitude, double Longitude, string AdresseTrouvee, bool AVerifier) : ResultatGeocodage;

    /// <summary>Le service a répondu, et n'a rien trouvé.</summary>
    public sealed record Introuvable : ResultatGeocodage;

    /// <summary>Tout le reste : injoignable, lent, refus, réponse illisible, disjoncteur ouvert.</summary>
    public sealed record Injoignable : ResultatGeocodage;
}

public sealed record ReglagesGeocodage(Uri Adresse, int DelaiSecondes);

/// <summary>
/// La localisation d'un site par la Géoplateforme de l'IGN (0.3.0, SPEC.md lot A3 points 9 et 10).
/// Ce qui part : la voie, le code postal et la commune — rien d'autre que <c>User-Agent</c> et
/// <c>Host</c>.
///
/// Classe concrète, sans interface (CLAUDE.md §11) : la série substitue le gestionnaire primaire du
/// client nommé <see cref="NomClient"/>.
///
/// ⚠️ Le client est verrouillé dans <see cref="Enregistrer"/>, et chaque verrou ferme une fuite que le
/// faux gestionnaire de la série ne verrait pas : redirection suivie, cookie renvoyé d'une entreprise à
/// l'autre, <c>traceparent</c> qui porterait l'identifiant de corrélation de KLEAN, URL — donc adresse —
/// dans le journal. <c>GeocodageTests</c> les vérifie sur le gestionnaire réel.
/// </summary>
public sealed class ServiceGeocodage
{
    public const string NomClient = "geocodage";
    public const string AdresseParDefaut = "https://data.geopf.fr/geocodage";
    public const int DelaiParDefaut = 5;

    /// <summary>Strictement sous les 30 s du délai de requête : sinon le jeton de la requête gagnerait, et un 504 remplacerait le 503.</summary>
    public const int DelaiMaximal = 20;

    public const int TailleMaximaleReponse = 64 * 1024;

    /// <summary>À éprouver sur le terrain avant d'en faire un réglage — la règle du rayon du passage (§9 ter).</summary>
    public const double SeuilScore = 0.7;

    public const int LongueurMaximaleAdresseTrouvee = 200;

    private static readonly TimeSpan PauseParDefaut = TimeSpan.FromSeconds(5);
    private static readonly TimeSpan PauseMaximale = TimeSpan.FromSeconds(30);

    private readonly IHttpClientFactory _fabrique;
    private readonly string _base;

    /// <summary>Le disjoncteur : avant cet instant (ticks UTC), aucun appel ne part.</summary>
    private long _reprise;

    public ServiceGeocodage(IHttpClientFactory fabrique, IConfiguration configuration)
    {
        _fabrique = fabrique;
        _base = Lire(configuration).Adresse.AbsoluteUri.TrimEnd('/');
    }

    /// <summary>
    /// Lit et valide la configuration. ⚠️ Appelée deux fois, et les deux sont nécessaires : par
    /// <see cref="Enregistrer"/>, ce qui fait refuser le démarrage ; et dans le délégué du client, pour
    /// la valeur finale de la configuration.
    /// </summary>
    public static ReglagesGeocodage Lire(IConfiguration configuration)
    {
        var brute = configuration["Geocodage:Adresse"];
        var texte = string.IsNullOrWhiteSpace(brute) ? AdresseParDefaut : brute.Trim();

        // La valeur n'est pas recopiée dans le message : elle pourrait porter des identifiants.
        if (!Uri.TryCreate(texte, UriKind.Absolute, out var adresse)
            || adresse.Scheme != Uri.UriSchemeHttps
            || adresse.UserInfo.Length > 0
            || adresse.Query.Length > 0
            || adresse.Fragment.Length > 0)
        {
            throw new InvalidOperationException(
                "Geocodage:Adresse doit être une adresse absolue en https, sans identifiants, requête ni "
                + $"fragment : configurée à moitié. Par défaut : {AdresseParDefaut}.");
        }

        var delai = DelaiParDefaut;
        var delaiBrut = configuration["Geocodage:DelaiSecondes"];

        if (!string.IsNullOrWhiteSpace(delaiBrut))
        {
            if (!int.TryParse(delaiBrut.Trim(), NumberStyles.None, CultureInfo.InvariantCulture, out delai)
                || delai < 1 || delai > DelaiMaximal)
            {
                throw new InvalidOperationException(
                    $"Geocodage:DelaiSecondes doit être un entier de 1 à {DelaiMaximal} : configuré à moitié. "
                    + $"Par défaut : {DelaiParDefaut}.");
            }
        }

        return new ReglagesGeocodage(adresse, delai);
    }

    /// <summary>Enregistre le service et son client nommé. Appelée par <c>Program.cs</c>, et par la série sur une collection neuve.</summary>
    public static void Enregistrer(IServiceCollection services, IConfiguration configuration)
    {
        Lire(configuration);

        services.AddSingleton<ServiceGeocodage>();

        services.AddHttpClient(NomClient, (fournisseur, client) =>
            {
                var reglages = Lire(fournisseur.GetRequiredService<IConfiguration>());
                client.Timeout = TimeSpan.FromSeconds(reglages.DelaiSecondes);
                client.MaxResponseContentBufferSize = TailleMaximaleReponse;
                client.DefaultRequestHeaders.UserAgent.ParseAdd("KLEAN");
            })
            .UseSocketsHttpHandler((gestionnaire, _) =>
            {
                gestionnaire.AllowAutoRedirect = false;
                gestionnaire.UseCookies = false;
                gestionnaire.ActivityHeadersPropagator = DistributedContextPropagator.CreateNoOutputPropagator();
                gestionnaire.AutomaticDecompression = DecompressionMethods.None;
            })
            // La garde principale du journal technique : un niveau Debug réglé par l'exploitant ne la
            // défait pas. Le niveau de appsettings.json en est la redondance (CLAUDE.md §10).
            .RemoveAllLoggers();
    }

    /// <summary>
    /// <paramref name="recherche"/> est déjà contrôlée par l'appelant : de 3 à 200 caractères, une
    /// lettre ou un chiffre en tête — les bornes mesurées du service. Le code postal n'est envoyé que
    /// s'il compte exactement cinq chiffres, sinon le service répondrait 400.
    /// </summary>
    public async Task<ResultatGeocodage> LocaliserAsync(string recherche, string? codePostal, CancellationToken annulation)
    {
        if (DateTime.UtcNow.Ticks < Interlocked.Read(ref _reprise)) return new ResultatGeocodage.Injoignable();

        // ⚠️ Concaténation et encodage un par un, jamais une combinaison d'Uri : celle-ci perdrait le
        // dernier segment d'une base sans « / » final, et ferait changer d'hôte une partie relative en « // ».
        var url = new StringBuilder(_base)
            .Append("/search?q=").Append(Uri.EscapeDataString(recherche));

        if (codePostal is { Length: 5 } && codePostal.All(char.IsAsciiDigit))
        {
            url.Append("&postcode=").Append(codePostal);
        }

        url.Append("&limit=1&index=address");

        try
        {
            // GetAsync en mode par défaut : le corps est mis en tampon, donc le plafond de 64 Kio et le
            // délai couvrent tout le corps — y compris une réponse servie au compte-gouttes.
            using var reponse = await _fabrique.CreateClient(NomClient).GetAsync(url.ToString(), annulation);

            if (reponse.StatusCode == HttpStatusCode.TooManyRequests)
            {
                Suspendre(reponse);
                return new ResultatGeocodage.Injoignable();
            }

            if (!reponse.IsSuccessStatusCode) return new ResultatGeocodage.Injoignable();

            return Analyser(await reponse.Content.ReadAsByteArrayAsync(annulation));
        }
        catch (OperationCanceledException) when (annulation.IsCancellationRequested)
        {
            throw;
        }
        catch (Exception)
        {
            // Dont le délai du client : une TaskCanceledException, mais avec un jeton de requête non annulé.
            return new ResultatGeocodage.Injoignable();
        }
    }

    /// <summary>Réarme le disjoncteur. Réservé à la série, qui partage une instance entre les tests.</summary>
    public void Rearmer() => Interlocked.Exchange(ref _reprise, 0);

    private void Suspendre(HttpResponseMessage reponse)
    {
        var entete = reponse.Headers.RetryAfter;
        var pause = entete?.Delta
            ?? (entete?.Date is { } date ? date - DateTimeOffset.UtcNow : PauseParDefaut);

        if (pause <= TimeSpan.Zero) pause = PauseParDefaut;
        if (pause > PauseMaximale) pause = PauseMaximale;

        Interlocked.Exchange(ref _reprise, (DateTime.UtcNow + pause).Ticks);
    }

    private static ResultatGeocodage Analyser(byte[] corps)
    {
        using var document = JsonDocument.Parse(corps);
        var features = document.RootElement.GetProperty("features");

        if (features.ValueKind != JsonValueKind.Array) return new ResultatGeocodage.Injoignable();
        if (features.GetArrayLength() == 0) return new ResultatGeocodage.Introuvable();

        var premier = features[0];
        var coordonnees = premier.GetProperty("geometry").GetProperty("coordinates");

        // GeoJSON : [longitude, latitude].
        if (coordonnees.ValueKind != JsonValueKind.Array || coordonnees.GetArrayLength() < 2
            || coordonnees[0].ValueKind != JsonValueKind.Number || coordonnees[1].ValueKind != JsonValueKind.Number
            || !coordonnees[0].TryGetDouble(out var longitude) || !coordonnees[1].TryGetDouble(out var latitude)
            || !double.IsFinite(latitude) || latitude < -90 || latitude > 90
            || !double.IsFinite(longitude) || longitude < -180 || longitude > 180)
        {
            return new ResultatGeocodage.Injoignable();
        }

        var proprietes = premier.GetProperty("properties");

        var label = proprietes.TryGetProperty("label", out var l) && l.ValueKind == JsonValueKind.String
            ? Nettoyer(l.GetString()!)
            : "";

        if (label.Length == 0) return new ResultatGeocodage.Injoignable();

        double? score = proprietes.TryGetProperty("score", out var s) && s.ValueKind == JsonValueKind.Number
            && s.TryGetDouble(out var valeur) ? valeur : null;

        var type = proprietes.TryGetProperty("type", out var t) && t.ValueKind == JsonValueKind.String
            ? t.GetString()
            : null;

        // Un score absent, une adresse placée au centre de sa rue ou de sa commune : à vérifier.
        var aVerifier = !(score >= SeuilScore) || type != "housenumber";

        return new ResultatGeocodage.Trouve(latitude, longitude, label, aVerifier);
    }

    /// <summary>
    /// Un texte d'un tiers, renvoyé à l'écran : sans caractère de contrôle ni de bidirectionnalité —
    /// qui pourraient faire afficher une autre adresse que celle trouvée —, et borné.
    /// </summary>
    private static string Nettoyer(string brut)
    {
        var propre = new string(brut.Where(c => !char.IsControl(c)
            && c is not ('\u200E' or '\u200F' or '\u061C')
            && c is not (>= '\u202A' and <= '\u202E')
            && c is not (>= '\u2066' and <= '\u2069')).ToArray()).Trim();

        return propre.Length > LongueurMaximaleAdresseTrouvee ? propre[..LongueurMaximaleAdresseTrouvee] : propre;
    }
}
```

```csharp
    /// <summary>
    /// Localise un site par la Géoplateforme de l'IGN (0.3.0). Administrateur seul, comme l'écriture des
    /// sites ; l'adresse voyage dans le <b>corps</b> — un <c>GET ?adresse=</c> la ferait entrer dans le
    /// journal des requêtes. Rien n'est enregistré : l'écran remplit ses champs, le patron relit puis
    /// enregistre.
    /// </summary>
    [Authorize(Roles = RolesKlean.Administrateur)]
    [HttpPost("localiser")]
    public async Task<ActionResult<LocalisationDto>> Localiser(LocaliserSiteDto dto, CancellationToken annulation)
    {
        var compte = User.FindFirstValue(ClaimTypes.NameIdentifier) ?? "";

        if (!_debit.Autoriser($"localisation:compte:{compte}", PlafondsDebit.LocalisationsParCompte,
                PlafondsDebit.FenetreLocalisations, out var attente))
        {
            return this.TropDeTentatives(attente, PlafondsDebit.FenetreLocalisations);
        }

        if (!_debit.Autoriser("localisation:global", PlafondsDebit.LocalisationsGlobales,
                PlafondsDebit.FenetreLocalisationsGlobales, out var attenteGlobale))
        {
            return this.TropDeTentatives(attenteGlobale, PlafondsDebit.FenetreLocalisationsGlobales);
        }

        // L'entreprise du jeton, et elle seule : sous la RLS, et cadrée sur son Id. Introuvable vaut désactivé.
        var entrepriseId = _tenant.EntrepriseId;
        var actif = entrepriseId is not null && await _db.Entreprises.AsNoTracking()
            .Where(e => e.Id == entrepriseId)
            .Select(e => e.GeocodageActif)
            .FirstOrDefaultAsync(annulation);

        if (!actif) return Conflict(RefusDesactive);

        var adresse = (dto.Adresse ?? "").Trim();
        var codePostal = (dto.CodePostal ?? "").Trim();
        var ville = (dto.Ville ?? "").Trim();

        if (adresse.Length == 0) return BadRequest(RefusAdresseVide);
        if (adresse.Length > 200) return BadRequest("L’adresse dépasse 200 caractères.");
        if (codePostal.Length > 10) return BadRequest("Le code postal dépasse 10 caractères.");
        if (ville.Length > 100) return BadRequest("La ville dépasse 100 caractères.");
        if (new[] { adresse, codePostal, ville }.Any(texte => texte.Any(char.IsControl))) return BadRequest(RefusInvisible);

        // Les bornes mesurées du service : de 3 à 200 caractères, une lettre ou un chiffre en tête.
        var recherche = ville.Length == 0 ? adresse : $"{adresse} {ville}";
        if (recherche.Length > 200) return BadRequest(RefusTropLongue);
        if (recherche.Length < 3 || !char.IsLetterOrDigit(recherche[0])) return BadRequest(RefusDebut);

        return await _geocodage.LocaliserAsync(recherche, codePostal, annulation) switch
        {
            ResultatGeocodage.Trouve trouve => Ok(new LocalisationDto
            {
                Latitude = trouve.Latitude,
                Longitude = trouve.Longitude,
                AdresseTrouvee = trouve.AdresseTrouvee,
                AVerifier = trouve.AVerifier
            }),
            ResultatGeocodage.Introuvable => BadRequest(RefusIntrouvable),
            _ => StatusCode(StatusCodes.Status503ServiceUnavailable, RefusInjoignable)
        };
    }
```

```ts
/**
 * La localisation d'un site (0.3.0) : ce qui se décide hors du composant — le refus local, le corps
 * envoyé, la chaîne d'une coordonnée, les phrases du résultat et d'un refus, la lecture du réglage.
 *
 * ⚠️ Pur, et sans import : chargé par `node --test` (CLAUDE.md §10). Les refus recopient ceux de
 * `SitesController.cs`, que la série lit : un refus local épargne un appel que le serveur refuserait,
 * et il doit le refuser dans les mêmes mots.
 */

export interface CorpsLocalisation {
  adresse: string
  codePostal: string | null
  ville: string | null
}

export interface ResultatLocalisation {
  latitude: number
  longitude: number
  adresseTrouvee: string
  aVerifier: boolean
}

export interface Bornes {
  adresse: number
  codePostal: number
  ville: number
}

/** Ce qu'il faut de la description : le nom des champs et leurs `longueurMax`. */
interface ChampBorne {
  nom: string
  longueurMax?: number
}

interface NomsLocalisation {
  adresse: string
  codePostal: string
  ville: string
}

export const AideLocaliser =
  'Envoie la voie, le code postal et la commune à l’IGN. Pour le domicile d’un particulier, c’est une '
  + 'donnée personnelle : mentionnez-le dans l’information que vous lui donnez.'

export const MotAVerifier = 'à vérifier : le point est approximatif'

/** La borne de la recherche envoyée, mesurée sur le service : de 3 à 200 caractères. */
const LongueurRecherche = 200

export const Refus = {
  adresseVide: 'Saisissez l’adresse du site avant de le localiser.',
  invisible: 'L’adresse, le code postal ou la ville contient un caractère invisible : retapez-les.',
  tropLongue: 'L’adresse et la ville dépassent ensemble 200 caractères : abrégez-les.',
  debut: 'L’adresse doit commencer par un numéro ou une lettre, et compter au moins trois caractères.',
} as const

// eslint-disable-next-line no-control-regex
const Controle = /[\u0000-\u001F\u007F-\u009F]/

export function bornesDe(champs: readonly ChampBorne[], noms: NomsLocalisation): Bornes {
  const borne = (nom: string): number => {
    const longueur = champs.find((champ) => champ.nom === nom)?.longueurMax
    if (longueur === undefined) throw new Error(`Le champ ${nom} n’a pas de longueurMax.`)
    return longueur
  }

  return { adresse: borne(noms.adresse), codePostal: borne(noms.codePostal), ville: borne(noms.ville) }
}

function recherche(adresse: string, ville: string): string {
  return ville === '' ? adresse : `${adresse} ${ville}`
}

export function refusLocal(adresse: string, codePostal: string, ville: string, bornes: Bornes): string | null {
  const a = adresse.trim()
  const c = codePostal.trim()
  const v = ville.trim()

  if (a === '') return Refus.adresseVide
  if (a.length > bornes.adresse) return `L’adresse dépasse ${bornes.adresse} caractères.`
  if (c.length > bornes.codePostal) return `Le code postal dépasse ${bornes.codePostal} caractères.`
  if (v.length > bornes.ville) return `La ville dépasse ${bornes.ville} caractères.`
  if ([a, c, v].some((texte) => Controle.test(texte))) return Refus.invisible

  const envoyee = recherche(a, v)
  if (envoyee.length > LongueurRecherche) return Refus.tropLongue
  if (envoyee.length < 3 || !/^[\p{L}\p{N}]/u.test(envoyee)) return Refus.debut

  return null
}

export function corpsLocalisation(adresse: string, codePostal: string, ville: string): CorpsLocalisation {
  const c = codePostal.trim()
  const v = ville.trim()
  return { adresse: adresse.trim(), codePostal: c === '' ? null : c, ville: v === '' ? null : v }
}

/**
 * En chaîne, arrondie à six décimales — le `pas` des champs : les champs nombre et `corpsDe` ne lisent
 * que des chaînes, et un double non arrondi ferait refuser l'enregistrement par le navigateur.
 */
export function chaineCoordonnee(valeur: number): string {
  const arrondie = Number(valeur.toFixed(6))
  return arrondie === 0 ? '0' : String(arrondie)
}

export function phraseTrouvee(resultat: ResultatLocalisation): string {
  return `Latitude et longitude renseignées. Trouvé : ${resultat.adresseTrouvee}`
}

/** Le texte d'un refus ; `null` pour une annulation, qui n'est pas un échec mais le clic suivant. */
export function messageRefus(echec: unknown): string | null {
  if (echec instanceof Error) {
    if (echec.name === 'AbortError') return null
    if (echec.message !== '') return echec.message
  }
  return 'La localisation a échoué.'
}

export function reglageActif(page: unknown): boolean {
  if (typeof page !== 'object' || page === null) return false
  const elements = (page as { elements?: unknown }).elements
  if (!Array.isArray(elements) || elements.length === 0) return false
  const premiere = elements[0] as { geocodageActif?: unknown } | null
  return premiere?.geocodageActif === true
}

export function routeLocalisation(route: string): string {
  return `${route}/localiser`
}
```

```tsx
interface ProprietesLocalisation {
  etat: EtatLocalisation
  onLocaliser: () => void
}

/**
 * Le bouton et le résultat. ⚠️ Jamais `disabled` pendant l'appel : le navigateur ferait perdre le focus
 * au bouton. Le résultat vit dans un `<output>` toujours rendu — un conteneur vivant doit exister avant
 * que son contenu change, sinon le lecteur d'écran ne l'annonce pas.
 */
function Localisation({ etat, onLocaliser }: ProprietesLocalisation) {
  const enCours = etat.etat === 'enCours'

  return (
    <div className="localisation">
      <button
        className="localisation__bouton"
        type="button"
        aria-disabled={enCours}
        aria-describedby="localisation-aide"
        onClick={onLocaliser}
      >
        {enCours ? 'Localisation…' : 'Localiser'}
      </button>

      <p className="champ__aide" id="localisation-aide">
        {AideLocaliser}
      </p>

      <output className="localisation__resultat">
        {etat.etat === 'trouvee' && (
          <>
            {phraseTrouvee(etat.resultat)}
            {etat.resultat.aVerifier && (
              <>
                {' — '}
                <strong className="localisation__a-verifier">{MotAVerifier}</strong>
              </>
            )}
          </>
        )}
      </output>

      {etat.etat === 'refusee' && (
        <p className="formulaire__erreur" role="alert">
          <strong className="formulaire__erreur-mot">Refusé —</strong> {etat.message}
        </p>
      )}
    </div>
  )
}
```
